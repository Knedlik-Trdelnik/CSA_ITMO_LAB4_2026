цело a_high = 0;
цело a_low = 2000000000;

цело b_high = 0;
цело b_low = 2000000000;

цело res_high = 0;
цело res_low = 0;

; Для проверки переноса мы используем деление (сдвиг).
; Так как у вас есть операция деления ":", мы можем эмулировать переполнение.
; В 32-битном знаковом числе максимум это 2147483647.
; Мы проверяем, не превысит ли сумма это число.

цело limit = 2147483647;
цело diff = 0;

main;
di;

; Пытаемся сложить младшие части
diff = limit - a_low;

if ( b_low > diff );
    ; Произошло переполнение!
    ; Увеличиваем старшую часть на 1
    res_high = a_high + b_high + 1;

    ; Высчитываем, сколько "осталось" в младшей части
    ; В реальности это сложная битовая магия, но для алгоритмики достаточно показать логику:
    res_low = b_low - diff - 1;
if-end;

if ( diff > b_low );
    ; Переполнения нет
    res_high = a_high + b_high;
    res_low = a_low + b_low;
if-end;

if ( diff == b_low );
    ; Переполнения нет
    res_high = a_high + b_high;
    res_low = a_low + b_low;
if-end;

; Выводим результат (должно получиться res_high=1, res_low=1852516352)
; Так как у нас нет print_num, выведем в порты буквы как "ОК" или просто оставим в памяти.
; Особенность вашего варианта - результат вычислений проверяется по дампу памяти или журналу.
; Поэтому мы просто выведем букву "D" (Done)

пиши_память( 68 ); ; 68 = 'D'
пиши_память( 10 );
main-end;
